class AuthGate extends StatelessWidget {
  const AuthGate({super.key});

  Future<Map<String, dynamic>> _loadProfile(User user) async {
    final snapshot = await FirebaseFirestore.instance
        .collection('users')
        .doc(user.uid)
        .get();

    if (!snapshot.exists) {
      throw Exception(
        'Your account profile is missing. '
        'Please contact support before continuing.',
      );
    }

    final data = snapshot.data();

    if (data == null ||
        (data['role'] != 'parent' && data['role'] != 'child')) {
      throw Exception(
        'Your account has an invalid role. '
        'Please contact support.',
      );
    }

    return data;
  }

  @override
  Widget build(BuildContext context) {
    return StreamBuilder<User?>(
      stream: FirebaseAuth.instance.authStateChanges(),
      builder: (context, authSnapshot) {
        if (authSnapshot.connectionState == ConnectionState.waiting) {
          return const _LoadingScreen();
        }

        final user = authSnapshot.data;

        if (user == null) {
          return const AuthScreen();
        }

        return FutureBuilder<Map<String, dynamic>>(
          key: ValueKey(user.uid),
          future: _loadProfile(user),
          builder: (context, profileSnapshot) {
            if (profileSnapshot.connectionState ==
                ConnectionState.waiting) {
              return const _LoadingScreen();
            }

            if (profileSnapshot.hasError) {
              return _MessageScreen(
                message: 'Could not load your account profile.\n\n'
                    '${profileSnapshot.error}',
                onSignOut: () async {
                  await FirebaseAuth.instance.signOut();
                },
              );
            }

            final profile = profileSnapshot.data!;
            final role = profile['role'];

            if (role == 'child') {
              final accountType = profile['accountType'];

              final parentUid = profile['parentUid'] as String?;
              final childProfileId =
                  profile['childProfileId'] as String?;

              // A parent-managed child must have both linking fields.
              if (accountType == 'parentManagedChild' &&
                  (parentUid == null ||
                      parentUid.isEmpty ||
                      childProfileId == null ||
                      childProfileId.isEmpty)) {
                return _MessageScreen(
                  message: 'Your child account is not properly linked '
                      'to its parent profile. Please contact support.',
                  onSignOut: () async {
                    await FirebaseAuth.instance.signOut();
                  },
                );
              }

              return KidsModeScreen(
                onExitKidsMode: () async {
                  await FirebaseAuth.instance.signOut();
                },
                parentUid: parentUid,
                childProfileId: childProfileId,
              );
            }

            return const MainNavigation();
          },
        );
      },
    );
  }
}

class MainNavigation extends StatefulWidget {
  const MainNavigation({super.key});

  @override
  State<MainNavigation> createState() => _MainNavigationState();
}

class _MainNavigationState extends State<MainNavigation> {
  static const String _kidsModeKey = 'kids_mode_enabled';

  int currentIndex = 0;
  bool kidsModeEnabled = false;
  bool preferencesLoaded = false;

  // The selected child belonging to the signed-in parent.
  String? selectedChildParentUid;
  String? selectedChildProfileId;

  static const adultDestinations = [
    RosaryDestination(Icons.home_rounded, 'Home'),
    RosaryDestination(Icons.auto_stories_rounded, 'Learn'),
    RosaryDestination(Icons.favorite_rounded, 'Pray'),
    RosaryDestination(Icons.insights_rounded, 'Progress'),
    RosaryDestination(Icons.person_rounded, 'Profile'),
  ];

  @override
  void initState() {
    super.initState();
    _loadKidsMode();
  }

  Future<void> _loadKidsMode() async {
    final preferences = await SharedPreferences.getInstance();

    if (!mounted) return;

    // Do not automatically reopen a parent's Kids Mode without
    // knowing which child was selected.
    final savedKidsMode = preferences.getBool(_kidsModeKey) ?? false;

    setState(() {
      kidsModeEnabled = false;
      preferencesLoaded = true;
    });

    if (savedKidsMode) {
      await preferences.setBool(_kidsModeKey, false);
    }
  }

  Future<void> _setKidsMode(bool enabled) async {
    if (enabled) {
      // A parent must select a child before entering Kids Mode.
      if (selectedChildParentUid == null ||
          selectedChildProfileId == null) {
        return;
      }
    }

    final preferences = await SharedPreferences.getInstance();
    await preferences.setBool(_kidsModeKey, enabled);

    if (!mounted) return;

    setState(() {
      kidsModeEnabled = enabled;
      currentIndex = 0;

      if (!enabled) {
        selectedChildParentUid = null;
        selectedChildProfileId = null;
      }
    });
  }

  // Called by ProfileScreen after the parent selects a child.
  Future<void> _enterKidsModeForChild(String childProfileId) async {
    final user = FirebaseAuth.instance.currentUser;

    if (user == null) return;

    selectedChildParentUid = user.uid;
    selectedChildProfileId = childProfileId;

    await _setKidsMode(true);
  }

  void _navigateToTab(int index) {
    if (index < 0 || index >= adultDestinations.length) return;

    setState(() {
      currentIndex = index;
    });
  }

  @override
  Widget build(BuildContext context) {
    if (!preferencesLoaded) {
      return const Scaffold(
        body: Center(
          child: CircularProgressIndicator(),
        ),
      );
    }

    if (kidsModeEnabled) {
      return KidsModeScreen(
        onExitKidsMode: () => _setKidsMode(false),
        parentUid: selectedChildParentUid,
        childProfileId: selectedChildProfileId,
      );
    }

    final screens = <Widget>[
      HomeScreen(onNavigate: _navigateToTab),
      const LearnScreen(),
      const PrayScreen(),
      const ProgressScreen(),
      ProfileScreen(
        kidsModeEnabled: kidsModeEnabled,
        onKidsModeChanged: _setKidsMode,
        onChildSelected: _enterKidsModeForChild,
      ),
    ];

    return Scaffold(
      extendBody: true,
      body: IndexedStack(
        index: currentIndex,
        children: screens,
      ),
      bottomNavigationBar: LiquidNavigationBar(
        currentIndex: currentIndex,
        destinations: adultDestinations,
        onChanged: _navigateToTab,
      ),
    );
  }
}

  String _normalizeUsername(String value) {
    return value.trim().toLowerCase().replaceAll('@smartrosary.app', '');
  }

  bool _isValidUsername(String value) {
    return value.length >= 3 &&
        RegExp(r'^[a-z0-9._]+$').hasMatch(value);
  }

  String _getLoginEmail(String value) {
    final input = value.trim().toLowerCase();

    if (input.contains('@')) {
      return input;
    }

    return '${_normalizeUsername(input)}@smartrosary.app';
  }

  Future<void> _submit() async {
    if (!_formKey.currentState!.validate()) return;

    setState(() => _loading = true);

    try {
      final auth = FirebaseAuth.instance;
      final firestore = FirebaseFirestore.instance;

      final input = _email.text.trim();
      final password = _password.text;

      final loginEmail = _getLoginEmail(input);

      if (_registering) {
        final name = _name.text.trim();

        final credential = await auth.createUserWithEmailAndPassword(
          email: loginEmail,
          password: password,
        );

        final user = credential.user!;

        final profileData = <String, dynamic>{
          'uid': user.uid,
          'name': name,
          'email': loginEmail,
          'role': _role,
          'accountType': _role == 'parent'
              ? 'parent'
              : 'independentChild',
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        };

        if (_role == 'child') {
          profileData['username'] = _normalizeUsername(input);
        }

        await firestore
            .collection('users')
            .doc(user.uid)
            .set(profileData);

        await user.updateDisplayName(name);
      } else {
        await auth.signInWithEmailAndPassword(
          email: loginEmail,
          password: password,
        );
      }
    } on FirebaseAuthException catch (e) {
      _message(switch (e.code) {
        'invalid-email' =>
          'Enter a valid email address or username.',
        'user-not-found' =>
          'No account found. Check your email or username.',
        'wrong-password' || 'invalid-credential' =>
          'Email/username or password is incorrect.',
        'email-already-in-use' =>
          'That email or username already has an account.',
        'weak-password' =>
          'Use a stronger password (at least 6 characters).',
        'too-many-requests' =>
          'Too many attempts. Try again later.',
        'network-request-failed' =>
          'Check your internet connection.',
        'operation-not-allowed' =>
          'Email/password authentication is not enabled in Firebase.',
        _ => e.message ?? 'Authentication failed. Please try again.',
      });
    } catch (e) {
      _message(
        'Could not complete the request. Check your connection and try again.',
      );
    } finally {
      if (mounted) {
        setState(() => _loading = false);
      }
    }
  }

      _message(
        'Enter the real email address associated with your account.',
      );
      return;

    try {
      await FirebaseAuth.instance.sendPasswordResetEmail(
        email: input.toLowerCase(),
      );

      _message(
        'If an account exists for that email, a reset email will be sent.',
      );
    } on FirebaseAuthException catch (e) {
      _message(e.message ?? 'Could not send the reset email.');
    }
  }

                                .set({
                              'name': name,
                              'email': user.email,
                              'updatedAt': FieldValue.serverTimestamp(),
                            }, SetOptions(merge: true));

                            await user.updateDisplayName(name);

                            if (dialogContext.mounted) {
                              Navigator.pop(dialogContext, name);
                            }
                          } catch (error) {
                            setDialogState(() => saving = false);

                            if (dialogContext.mounted) {
                              ScaffoldMessenger.of(dialogContext).showSnackBar(
                                SnackBar(
                                  content: Text(
                                    'Could not update profile: $error',
                                  ),
                                ),
                              );
                            }
                          }
                        },
                  style: FilledButton.styleFrom(backgroundColor: purple),
                  child: saving
                      ? const SizedBox(
                          height: 18,
                          width: 18,
                          child: CircularProgressIndicator(
                            strokeWidth: 2,
                            color: Colors.white,
                          ),
                        )
                      : const Text('Save changes'),
                ),
              ],
            );
          },
        );
      },
    );

    controller.dispose();

    if (newName != null && context.mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Your profile was updated successfully.'),
        ),
      );
    }
  }

  // ------------------------------------------------------------
  // CREATE CHILD FIREBASE ACCOUNT
  // ------------------------------------------------------------

  Future<void> _registerChild(BuildContext context) async {
    final parent = FirebaseAuth.instance.currentUser;

    if (parent == null) return;

    final nameController = TextEditingController();
    final usernameController = TextEditingController();
    final passwordController = TextEditingController();

    final formKey = GlobalKey<FormState>();

    bool obscurePassword = true;
    bool saving = false;

    final credentials = await showDialog<Map<String, String>>(
      context: context,
      barrierDismissible: false,
      builder: (dialogContext) {
        return StatefulBuilder(
          builder: (context, setDialogState) {
            return AlertDialog(
              title: const Text('Register a child'),
              content: SingleChildScrollView(
                child: Form(
                  key: formKey,
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      const Text(
                        'Create a personal Smart Rosary account for your child. '
                        'They can later sign in using their username and password.',
                        style: TextStyle(
                          color: muted,
                          fontSize: 12,
                          height: 1.5,
                        ),
                      ),
                      const SizedBox(height: 18),
                      TextFormField(
                        controller: nameController,
                        textCapitalization: TextCapitalization.words,
                        decoration: const InputDecoration(
                          labelText: 'Child name',
                          prefixIcon: Icon(Icons.child_care_rounded),
                          border: OutlineInputBorder(),
                        ),
                        validator: (value) {
                          if (value == null || value.trim().length < 2) {
                            return 'Enter a valid child name.';
                          }
                          return null;
                        },
                      ),
                      const SizedBox(height: 13),
                      TextFormField(
                        controller: usernameController,
                        autocorrect: false,
                        decoration: const InputDecoration(
                          labelText: 'Username',
                          hintText: 'e.g. brian_kid',
                          prefixIcon: Icon(Icons.person_outline_rounded),
                          border: OutlineInputBorder(),
                        ),
                        validator: (value) {
                          final username =
                              value?.trim().toLowerCase() ?? '';

                          if (username.length < 3) {
                            return 'Username must have at least 3 characters.';
                          }

                          if (!RegExp(r'^[a-z0-9._]+$')
                              .hasMatch(username)) {
                            return 'Use letters, numbers, dots or underscores.';
                          }

                          return null;
                        },
                      ),
                      const SizedBox(height: 13),
                      TextFormField(
                        controller: passwordController,
                        obscureText: obscurePassword,
                        decoration: InputDecoration(
                          labelText: 'Password',
                          prefixIcon: const Icon(Icons.lock_outline_rounded),
                          border: const OutlineInputBorder(),
                          suffixIcon: IconButton(
                            onPressed: () {
                              setDialogState(() {
                                obscurePassword = !obscurePassword;
                              });
                            },
                            icon: Icon(
                              obscurePassword
                                  ? Icons.visibility_outlined
                                  : Icons.visibility_off_outlined,
                            ),
                          ),
                        ),
                        validator: (value) {
                          if (value == null || value.length < 6) {
                            return 'Password must have at least 6 characters.';
                          }
                          return null;
                        },
                      ),
                    ],
                  ),
                ),
              ),
              actions: [
                TextButton(
                  onPressed: saving
                      ? null
                      : () => Navigator.pop(dialogContext),
                  child: const Text('Cancel'),
                ),
                FilledButton(
                  onPressed: saving
                      ? null
                      : () async {
                          if (!formKey.currentState!.validate()) return;

                          setDialogState(() => saving = true);

                          final name = nameController.text.trim();
                          final username =
                              usernameController.text.trim().toLowerCase();
                          final password = passwordController.text;

                          final loginEmail = '$username@smartrosary.app';

                          UserCredential? childCredential;
                          FirebaseAuth? childAuth;

                          try {
                            // Create or reuse a secondary Firebase app.
                            // This keeps the parent signed in on the
                            // default Firebase app.
                            FirebaseApp childApp;

                            try {
                              childApp =
                                  Firebase.app('childRegistration');
                            } catch (_) {
                              childApp = await Firebase.initializeApp(
                                name: 'childRegistration',
                                options: Firebase.app().options,
                              );
                            }

                            childAuth =
                                FirebaseAuth.instanceFor(app: childApp);

                            // Create the child's Firebase Auth account.
                            childCredential =
                                await childAuth.createUserWithEmailAndPassword(
                              email: loginEmail,
                              password: password,
                            );

                            final childUid = childCredential.user!.uid;

                            final firestore = FirebaseFirestore.instance;
                            final batch = firestore.batch();

                            // Parent-managed child profile.
                            final childProfileRef = firestore
                                .collection('users')
                                .doc(parent.uid)
                                .collection('kids')
                                .doc(childUid);

                            // Child's own Firebase user document.
                            final childUserRef = firestore
                                .collection('users')
                                .doc(childUid);

                            batch.set(childProfileRef, {
                              'childUid': childUid,
                              'parentUid': parent.uid,
                              'name': name,
                              'username': username,
                              'loginEmail': loginEmail,
                              'accountType': 'parentManaged',
                              'createdAt': FieldValue.serverTimestamp(),
                              'updatedAt': FieldValue.serverTimestamp(),
                            });

                            // IMPORTANT:
                            // uid and role are required by Firestore rules.
                            batch.set(childUserRef, {
                              'uid': childUid,
                              'name': name,
                              'username': username,
                              'email': loginEmail,
                              'role': 'child',
                              'parentUid': parent.uid,
                              'childProfileId': childUid,
                              'accountType': 'parentManagedChild',
                              'createdAt': FieldValue.serverTimestamp(),
                              'updatedAt': FieldValue.serverTimestamp(),
                            });

                            // Default Firestore instance remains
                            // authenticated as the parent.
                            await batch.commit();

                            // Sign out only the secondary child Auth.
                            await childAuth.signOut();

                            if (dialogContext.mounted) {
                              Navigator.pop(dialogContext, {
                                'name': name,
                                'username': username,
                                'password': password,
                              });
                            }
                          } catch (error, stackTrace) {
                            debugPrint(
                              'CHILD REGISTRATION ERROR: $error',
                            );
                            debugPrintStack(stackTrace: stackTrace);

                            // Best-effort cleanup of a newly created
                            // authentication account.
                            if (childCredential?.user != null) {
                              try {
                                await childCredential!.user!.delete();
                              } catch (cleanupError) {
                                debugPrint(
                                  'CHILD ACCOUNT CLEANUP ERROR: '
                                  '$cleanupError',
                                );
                              }
                            }

                            // Keep the parent registration dialog open.
                            if (dialogContext.mounted) {
                              setDialogState(() => saving = false);

                              ScaffoldMessenger.of(dialogContext).showSnackBar(
                                SnackBar(
                                  content: Text(
                                    'Could not register child: $error',
                                  ),
                                ),
                              );
                            }
                          }
                        },
                  style: FilledButton.styleFrom(backgroundColor: purple),
                  child: saving
                      ? const SizedBox(
                          height: 18,
                          width: 18,
                          child: CircularProgressIndicator(
                            strokeWidth: 2,
                            color: Colors.white,
                          ),
                        )
                      : const Text('Create account'),
                ),
              ],
            );
          },
        );
      },
    );

    nameController.dispose();
    usernameController.dispose();
    passwordController.dispose();

    if (credentials == null || !context.mounted) return;

    await showDialog<void>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Row(
          children: [
            Icon(Icons.check_circle_rounded, color: Colors.green),
            SizedBox(width: 9),
            Expanded(child: Text('Child account created')),
          ],
        ),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              '${credentials['name']} can now sign in independently.',
              style: const TextStyle(fontSize: 13),
            ),
            const SizedBox(height: 18),
            _CredentialRow(
              label: 'Username',
              value: credentials['username']!,
            ),
            const SizedBox(height: 10),
            _CredentialRow(
              label: 'Password',
              value: credentials['password']!,
            ),
            const SizedBox(height: 14),
            const Text(
              'Save these login details securely. The password will not be '
              'displayed again.',
              style: TextStyle(
                color: muted,
                fontSize: 11,
                height: 1.5,
              ),
            ),
          ],
        ),
        actions: [
          FilledButton(
            onPressed: () => Navigator.pop(dialogContext),
            style: FilledButton.styleFrom(backgroundColor: purple),
            child: const Text('Done'),
          ),
        ],
      ),
    );
  }

  Future<void> _chooseChild(BuildContext context) async {
    final parent = FirebaseAuth.instance.currentUser;

    if (parent == null) return;

    final kidsRef = FirebaseFirestore.instance
        .collection('users')
        .doc(parent.uid)
        .collection('kids');

    await showDialog<void>(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          title: const Text('Choose a child'),
          content: SizedBox(
            width: double.maxFinite,
            child: StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
              stream: kidsRef.orderBy('createdAt').snapshots(),
              builder: (context, snapshot) {
                if (snapshot.hasError) {
                  return const Text(
                    'Could not load children. Check your Firestore rules.',
                    style: TextStyle(color: Colors.red),
                  );
                }

                if (snapshot.connectionState == ConnectionState.waiting) {
                  return const Padding(
                    padding: EdgeInsets.all(20),
                    child: Center(child: CircularProgressIndicator()),
                  );
                }

                final children = snapshot.data?.docs ?? [];

                if (children.isEmpty) {
                  return const Padding(
                    padding: EdgeInsets.symmetric(vertical: 12),
                    child: Text(
                      'You have not registered any children yet. '
                      'Add a child first to use Kids Mode.',
                      style: TextStyle(
                        color: muted,
                        fontSize: 13,
                        height: 1.5,
                      ),
                    ),
                  );
                }

                return ListView.separated(
                  shrinkWrap: true,
                  itemCount: children.length,
                  separatorBuilder: (_, __) => const Divider(height: 1),
                  itemBuilder: (context, index) {
                    final child = children[index];
                    final data = child.data();

                    final name = (data['name'] ?? 'Child').toString();
                    final username =
                        (data['username'] ?? '').toString();

                    return ListTile(
                      contentPadding: EdgeInsets.zero,
                      leading: const CircleAvatar(
                        backgroundColor: Color(0xFFFFE7A8),
                        child: Icon(
                          Icons.child_care_rounded,
                          color: Color(0xFFB77916),
                        ),
                      ),
                      title: Text(
                        name,
                        style: const TextStyle(
                          fontWeight: FontWeight.w800,
                          color: ink,
                        ),
                      ),
                      subtitle: username.isEmpty ? null : Text('@$username'),
                      trailing: const Icon(
                        Icons.arrow_forward_ios_rounded,
                        size: 15,
                      ),
                      onTap: () async {
                        Navigator.pop(dialogContext);
                        await onChildSelected(child.id);
                      },
                    );
                  },
                );
              },
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Cancel'),
            ),
          ],
        );
      },
    );
  }

    final shouldSignOut = await showDialog<bool>(

        title: const Text('Sign out?'),
        content: const Text(

            style: FilledButton.styleFrom(backgroundColor: purple),

      if (!context.mounted) return;

        SnackBar(content: Text('Sign out failed: $error')),
      );
    }
  }

class _KidsModeScreenState extends State<KidsModeScreen> {
  int selectedTab = 0;

  KidsProgressService? progressService;
  bool loadingProgress = true;
  String? progressError;

  @override
  void initState() {
    super.initState();
    _initializeProgressService();
  }

  Future<void> _initializeProgressService() async {
    try {
      // CASE 1: Parent has selected a particular child.
      if (widget.parentUid != null &&
          widget.childProfileId != null) {
        progressService = KidsProgressService.parentManaged(
          parentUid: widget.parentUid!,
          childProfileId: widget.childProfileId!,
        );
      } else {
        // CASE 2: A child has signed in independently.
        final user = FirebaseAuth.instance.currentUser;

        if (user == null) {
          throw StateError(
            'No authenticated user for Kids Mode.',
          );
        }

        // Check whether this account was created by a parent.
        final userSnapshot = await FirebaseFirestore.instance
            .collection('users')
            .doc(user.uid)
            .get();

        final userData =
            userSnapshot.data() ?? <String, dynamic>{};

        final parentUid =
            userData['parentUid']?.toString();

        final childProfileId =
            userData['childProfileId']?.toString();

        if (parentUid != null &&
            parentUid.isNotEmpty &&
            childProfileId != null &&
            childProfileId.isNotEmpty) {
          // Parent-created child:
          // use the parent's child progress document.
          progressService = KidsProgressService.parentManaged(
            parentUid: parentUid,
            childProfileId: childProfileId,
          );
        } else {
          // Independent child:
          // use the child's own progress document.
          progressService =
              KidsProgressService.independent(user.uid);
        }
      }
    } catch (error) {
      progressError = error.toString();
    }

    if (!mounted) return;

    setState(() {
      loadingProgress = false;
    });
  }

class _StoryPreviewStrip extends StatelessWidget {
  const _StoryPreviewStrip();

  static const _tints = [
    _Palette.sky,
    _Palette.mint,
    _Palette.sun,
    _Palette.rose,
    _Palette.lavender,
  ];

  @override
  Widget build(BuildContext context) {
    final storiesQuery = FirebaseFirestore.instance
        .collection('kidsStories')
        .where('published', isEqualTo: true);

    return StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
      stream: storiesQuery.snapshots(),
      builder: (context, snapshot) {
        if (snapshot.hasError) {
          return Container(
            alignment: Alignment.center,
            padding: const EdgeInsets.all(16),
            decoration: BoxDecoration(
              color: Colors.white,
              borderRadius: BorderRadius.circular(22),
              border: Border.all(color: _Palette.border),
            ),
            child: const Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Icon(
                  Icons.cloud_off_rounded,
                  color: _Palette.purple,
                  size: 34,
                ),
                SizedBox(height: 8),
                Text(
                  'We could not load the stories.',
                  textAlign: TextAlign.center,
                  style: TextStyle(
                    color: _Palette.ink,
                    fontWeight: FontWeight.w800,
                  ),
                ),
              ],
            ),
          );
        }

        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(
            child: CircularProgressIndicator(
              color: _Palette.purple,
            ),
          );
        }

        final docs = [...(snapshot.data?.docs ?? [])];

        docs.sort((a, b) {
          final aOrder = a.data()['order'];
          final bOrder = b.data()['order'];

          final aValue = aOrder is num ? aOrder : 0;
          final bValue = bOrder is num ? bOrder : 0;

          return aValue.compareTo(bValue);
        });

        if (docs.isEmpty) {
          return Container(
            alignment: Alignment.center,
            padding: const EdgeInsets.all(18),
            decoration: BoxDecoration(
              color: Colors.white,
              borderRadius: BorderRadius.circular(22),
              border: Border.all(color: _Palette.border),
            ),
            child: const Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Icon(
                  Icons.auto_stories_rounded,
                  color: _Palette.purple,
                  size: 36,
                ),
                SizedBox(height: 8),
                Text(
                  'No published stories yet. Publish a story in the admin dashboard to see it here.',
                  textAlign: TextAlign.center,
                  style: TextStyle(
                    color: _Palette.muted,
                    fontWeight: FontWeight.w700,
                  ),
                ),
              ],
            ),
          );
        }

        return ListView.separated(
          scrollDirection: Axis.horizontal,
          itemCount: docs.length,
          separatorBuilder: (_, _) => const SizedBox(width: 12),
          itemBuilder: (context, index) {
            final data = docs[index].data();

            return _BibleStoryCard(
              title: (data['title'] ?? 'Untitled story').toString(),
              description: (data['description'] ?? '').toString(),
              imageUrl: (data['imageUrl'] ?? '').toString(),
              storyText: (data['storyText'] ?? '').toString(),
              activity: (data['activity'] ?? '').toString(),
              tint: _tints[index % _tints.length],
            );
          },
        );
      },
    );
  }
}

  void _updateActiveWord(Duration position) {
    _audioPosition = position;
    final prayer = _matchingPrayer(_currentPrayerDocs);
    final text = _displayText(prayer);
    final words = _words(text);
    if (words.isEmpty) return;

    int index;
    final timings = prayer?['wordTimings'] ?? prayer?['word_timestamps'];
    if (timings is List && timings.isNotEmpty) {
      index = -1;
      for (var i = 0; i < timings.length && i < words.length; i++) {
        final item = timings[i];
        if (item is Map) {
          final start = (item['start'] ?? item['startSeconds'] ?? 0) as num;
          final end = (item['end'] ?? item['endSeconds'] ?? start) as num;
          if (position.inMilliseconds >= start * 1000 &&
              position.inMilliseconds <= end * 1000) {
            index = i;
            break;
          }
        }
      }
      if (index < 0) {
        index = ((position.inMilliseconds / 1000) /
                ((_audioDuration?.inMilliseconds ?? 1) / 1000) * words.length)
            .floor()
            .clamp(0, words.length - 1).toInt();
      }
    } else {
      // Fallback estimate: evenly distribute words across the recording.
      // Add wordTimings to a prayer document for precise word-level alignment.
      final totalMs = _audioDuration?.inMilliseconds ?? 0;
      index = totalMs <= 0
          ? -1
          : ((position.inMilliseconds / totalMs) * words.length)
              .floor()
              .clamp(0, words.length - 1).toInt();
    }

    if (mounted && index != _activeWordIndex) {
      setState(() => _activeWordIndex = index);
    }
  }

  Map<String, dynamic>? _matchingPrayer(List<Map<String, dynamic>> prayers) {
    if (_step.prayerId.isEmpty) return null;
    final wanted = _step.prayerId.replaceAll(RegExp(r'[^a-z0-9]'), '').toLowerCase();
    final titleAliases = <String, Set<String>>{
      'signofcross': {'signofthecross', 'thesignofthecross'},
    };
    for (final prayer in prayers) {
      final title = (prayer['title'] ?? '')
          .toString()
          .toLowerCase()
          .replaceAll(RegExp(r'[^a-z0-9]'), '');
      final id = (prayer['_id'] ?? '').toString().toLowerCase();
      if (title == wanted ||
          (titleAliases[wanted]?.contains(title) ?? false) ||
          id == _step.prayerId) {
        return prayer;
      }
    }
    return null;
  }

  Future<void> _playPrayer(Map<String, dynamic>? prayer) async {
    final path = prayer?['audioPath'];
    if (path is! String || path.trim().isEmpty) {
      setState(() => _audioError = 'Audio has not been added to this prayer yet.');
      return;
    }
    setState(() {
      _requestingAudio = true;
      _audioError = null;
    });
    try {
      final user = FirebaseAuth.instance.currentUser;
      if (user == null) throw Exception('Please sign in again to play prayer audio.');
      final supabase = Supabase.instance.client;
      final token = await user.getIdToken();
      final response = await supabase.functions.invoke(
        'get-prayer-audio-url',
        headers: {'Authorization': 'Bearer ${token ?? ''}'},
        body: {'path': path},
      );
      final data = response.data;
      final result = data is Map<String, dynamic>
          ? data
          : data is Map
              ? Map<String, dynamic>.from(data)
              : <String, dynamic>{};
      final signedUrl = result['signedUrl'];
      if (signedUrl is! String || signedUrl.isEmpty) {
        throw Exception('The audio service did not return a playback URL.');
      }
      await _player.setUrl(signedUrl);
      _audioPosition = Duration.zero;
      _activeWordIndex = -1;
      await _player.play();
    } catch (error) {
      if (mounted) setState(() => _audioError = error.toString().replaceFirst('Exception: ', ''));
    } finally {
      if (mounted) setState(() => _requestingAudio = false);
    }
  }

  Future<void> _goNext() async {

      _showFinished();
    }
  }

      _audioError = null;
    });

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';

/// Progress is isolated by child identity.
///
/// Independent child:
/// users/{childUid}/kidsProgress/main
///
/// Parent-managed child:
/// users/{parentUid}/kids/{childProfileId}/progress/main
///
/// Guided sessions are stored below the selected progress document.
class KidsProgressService {
  KidsProgressService._(this._ref);

  final DocumentReference<Map<String, dynamic>> _ref;

  factory KidsProgressService.independent(String uid) {
    return KidsProgressService._(
      FirebaseFirestore.instance
          .collection('users')
          .doc(uid)
          .collection('kidsProgress')
          .doc('main'),
    );
  }

  factory KidsProgressService.parentManaged({
    required String parentUid,
    required String childProfileId,
  }) {
    return KidsProgressService._(
      FirebaseFirestore.instance
          .collection('users')
          .doc(parentUid)
          .collection('kids')
          .doc(childProfileId)
          .collection('progress')
          .doc('main'),
    );
  }

factory KidsProgressService.forChild({
  required String uid,
  String? parentUid,
  String? childProfileId,
}) {
  if (parentUid != null &&
      parentUid.isNotEmpty &&
      childProfileId != null &&
      childProfileId.isNotEmpty) {
    return KidsProgressService.parentManaged(
      parentUid: parentUid,
      childProfileId: childProfileId,
    );
  }

  return KidsProgressService.independent(uid);
}



  factory KidsProgressService.forCurrentUser() {
    final user = FirebaseAuth.instance.currentUser;

    if (user == null) {
      throw StateError('No authenticated user for Kids Mode.');
    }

    return KidsProgressService.independent(user.uid);
  }

  Stream<KidsProgress> watch() {
    return _ref.snapshots().map(KidsProgress.fromMap);
  }

  Future<void> completeLesson(String lessonId) {
    return _completeOnce(
      field: 'completedLessons',
      id: lessonId,
      awardStars: 5,
    );
  }

  Future<void> completePrayer(String prayerDocumentId) {
    return _completeOnce(
      field: 'completedPrayers',
      id: prayerDocumentId,
      awardStars: 5,
      markPrayerDay: true,
    );
  }

  /// Records newly listened seconds from the guided player.
  /// Set completed=true only when the activity actually finishes.
  Future<void> recordGuidedPlayback({
    required String prayerDocumentId,
    required int listenedSecondsDelta,
    int? durationSeconds,
    bool completed = false,
  }) async {
    if (prayerDocumentId.trim().isEmpty) {
      throw ArgumentError.value(
        prayerDocumentId,
        'prayerDocumentId',
        'Must be a Firestore prayers document ID.',
      );
    }

    final delta =
        listenedSecondsDelta < 0 ? 0 : listenedSecondsDelta;

    final sessionRef =
        _ref.collection('guidedSessions').doc(prayerDocumentId);

    await FirebaseFirestore.instance.runTransaction((transaction) async {
      final progressSnapshot = await transaction.get(_ref);
      final sessionSnapshot = await transaction.get(sessionRef);

      final progress =
          progressSnapshot.data() ?? <String, dynamic>{};
      final session =
          sessionSnapshot.data() ?? <String, dynamic>{};

      final completedPrayers = List<String>.from(
        progress['completedPrayers'] ?? const <String>[],
      );

      final alreadyCompleted =
          session['completed'] == true ||
          completedPrayers.contains(prayerDocumentId);

      final award = completed && !alreadyCompleted;

      final stars = progress['stars'] is num
          ? (progress['stars'] as num).toInt()
          : 0;

      final listened = session['listenedSeconds'] is num
          ? (session['listenedSeconds'] as num).toInt()
          : 0;

      final sessionUpdate = <String, dynamic>{
        'prayerId': prayerDocumentId,
        'listenedSeconds': listened + delta,
        'completed': completed || alreadyCompleted,
        'updatedAt': FieldValue.serverTimestamp(),
        'lastPlayedAt': FieldValue.serverTimestamp(),
      };

      if (durationSeconds != null && durationSeconds > 0) {
        sessionUpdate['durationSeconds'] = durationSeconds;
      }

      if (completed && !alreadyCompleted) {
        sessionUpdate['completedAt'] =
            FieldValue.serverTimestamp();
      }

      transaction.set(
        sessionRef,
        sessionUpdate,
        SetOptions(merge: true),
      );

      final progressUpdate = <String, dynamic>{
        'updatedAt': FieldValue.serverTimestamp(),
      };

      if (delta > 0 || completed) {
        progressUpdate['prayerDays'] =
            FieldValue.arrayUnion([_todayKey()]);
      }

      if (completed) {
        progressUpdate['completedPrayers'] =
            FieldValue.arrayUnion([prayerDocumentId]);
      }

      if (award) {
        progressUpdate['stars'] = stars + 5;
      }

      transaction.set(
        _ref,
        progressUpdate,
        SetOptions(merge: true),
      );
    });
  }

  Future<void> _completeOnce({
    required String field,
    required String id,
    required int awardStars,
    bool markPrayerDay = false,
  }) async {
    if (id.trim().isEmpty) {
      throw ArgumentError.value(
        id,
        'id',
        'Cannot be empty.',
      );
    }

    await FirebaseFirestore.instance.runTransaction((transaction) async {
      final snapshot = await transaction.get(_ref);
      final data = snapshot.data() ?? <String, dynamic>{};

      final completed = List<String>.from(
        data[field] ?? const <String>[],
      );

      if (completed.contains(id)) return;

      final stars = data['stars'] is num
          ? (data['stars'] as num).toInt()
          : 0;

      final update = <String, dynamic>{
        field: FieldValue.arrayUnion([id]),
        'stars': stars + awardStars,
        'updatedAt': FieldValue.serverTimestamp(),
      };

      if (markPrayerDay) {
        update['prayerDays'] =
            FieldValue.arrayUnion([_todayKey()]);
      }

      transaction.set(
        _ref,
        update,
        SetOptions(merge: true),
      );
    });
  }

  static String _todayKey() {
    final now = DateTime.now();

    return '${now.year.toString().padLeft(4, '0')}-'
        '${now.month.toString().padLeft(2, '0')}-'
        '${now.day.toString().padLeft(2, '0')}';
  }
}

class KidsProgress {
  final int stars;
  final Set<String> completedLessons;
  final Set<String> completedPrayers;
  final Set<String> prayerDays;

  const KidsProgress({
    this.stars = 0,
    this.completedLessons = const {},
    this.completedPrayers = const {},
    this.prayerDays = const {},
  });

  factory KidsProgress.fromMap(
    DocumentSnapshot<Map<String, dynamic>> snapshot,
  ) {
    return KidsProgress.fromData(
      snapshot.data() ?? const <String, dynamic>{},
    );
  }

  factory KidsProgress.fromData(Map<String, dynamic> data) {
    Set<String> readSet(String key) {
      final value = data[key];

      if (value is Iterable) {
        return value.whereType<String>().toSet();
      }

      return <String>{};
    }

    return KidsProgress(
      stars: data['stars'] is num
          ? (data['stars'] as num).toInt()
          : 0,
      completedLessons: readSet('completedLessons'),
      completedPrayers: readSet('completedPrayers'),
      prayerDays: readSet('prayerDays'),
    );
  }

  bool get firstPrayerBadge => completedPrayers.isNotEmpty;

  bool get maryBadge => completedPrayers.length >= 2;

  bool get rosaryBadge => completedPrayers.length >= 4;

  bool get peacefulHeartBadge =>
      completedLessons.length >= 4 &&
      completedPrayers.length >= 3;

  int get badges => [
        firstPrayerBadge,
        maryBadge,
        rosaryBadge,
        peacefulHeartBadge,
      ].where((earned) => earned).length;
}
